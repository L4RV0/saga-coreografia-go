// Package broker implementa el "Message Broker System" del video: un sistema
// de publicación/suscripción por tópicos. Los servicios NUNCA se llaman entre
// sí directamente; solo publican eventos y reaccionan a los eventos de otros.
//
// Esta versión vive en memoria (goroutines + canales) para que el proyecto
// corra sin instalar nada. Como los servicios dependen de la interfaz Broker,
// se puede cambiar por Kafka, RabbitMQ o NATS sin tocar su lógica.
package broker

import (
        "sync"

        "saga-coreografia/internal/events"
        "saga-coreografia/internal/logx"
)

// Handler procesa un evento recibido.
type Handler func(events.Event)

// Broker es lo único que los servicios conocen del sistema de mensajería.
type Broker interface {
        Publish(topic string, e events.Event)
        Subscribe(topic, subscriber string, h Handler)
}

type subscription struct {
        name string
        ch   chan events.Event
}

// InMemory es un broker asíncrono: Publish no espera a que los consumidores
// procesen el evento (igual que un broker real).
type InMemory struct {
        mu      sync.RWMutex
        subs    map[string][]*subscription
        pending sync.WaitGroup // eventos publicados que aún no terminan de procesarse
}

// NewInMemory crea un broker vacío.
func NewInMemory() *InMemory {
        return &InMemory{subs: make(map[string][]*subscription)}
}

// Subscribe registra un consumidor en un tópico. Cada suscripción tiene su
// propia cola (canal) y su propia goroutine, como un consumer group.
func (b *InMemory) Subscribe(topic, subscriber string, h Handler) {
        s := &subscription{name: subscriber, ch: make(chan events.Event, 256)}

        b.mu.Lock()
        b.subs[topic] = append(b.subs[topic], s)
        b.mu.Unlock()

        go func() {
                for e := range s.ch {
                        h(e)
                        b.pending.Done()
                }
        }()
}

// Publish entrega una copia del evento a cada suscriptor del tópico.
func (b *InMemory) Publish(topic string, e events.Event) {
        b.mu.RLock()
        defer b.mu.RUnlock()

        logx.Log("broker", logx.Gray, "📨 %-17s → %s", e.Type, topic)
        for _, s := range b.subs[topic] {
                b.pending.Add(1)
                s.ch <- e
        }
}

// WaitIdle bloquea hasta que todos los eventos publicados (y los que estos
// provocaron en cadena) fueron procesados. Sirve para la demo y los tests.
func (b *InMemory) WaitIdle() { b.pending.Wait() }

// Close cierra todas las colas.
func (b *InMemory) Close() {
        b.mu.Lock()
        defer b.mu.Unlock()
        for _, list := range b.subs {
                for _, s := range list {
                        close(s.ch)
                }
        }
        b.subs = map[string][]*subscription{}
}

// Package events define el "contrato" entre microservicios: los tópicos del
// broker y los eventos que viajan por ellos. Es lo ÚNICO que los servicios
// comparten; ninguno conoce el código ni la base de datos de otro.
package events

import (
        "crypto/rand"
        "encoding/hex"
        "time"
)

// Tópicos del broker (igual que en el video: order / inventory / payment topic).
// Cada servicio publica SOLO en su propio tópico.
const (
        TopicOrder     = "order-topic"
        TopicInventory = "inventory-topic"
        TopicPayment   = "payment-topic"
)

// Type es el nombre del evento (algo que YA ocurrió, por eso va en pasado).
type Type string

const (
        // Publicado por Order en order-topic.
        OrderCreated Type = "OrderCreated"

        // Publicados por Inventory en inventory-topic.
        InventoryReserved Type = "InventoryReserved" // transacción local T2 OK
        InventoryRejected Type = "InventoryRejected" // sin stock: T2 falló
        StockReleased     Type = "StockReleased"     // transacción COMPENSATORIA de T2

        // Publicados por Payment en payment-topic.
        PaymentProcessed Type = "PaymentProcessed" // transacción local T3 OK
        PaymentFailed    Type = "PaymentFailed"    // T3 falló -> dispara compensaciones
)

// Event es el mensaje que viaja por el broker.
type Event struct {
        ID         string // identificador único: permite descartar duplicados (idempotencia)
        Type       Type
        OrderID    string // "correlation id": une todos los eventos de una misma saga
        CustomerID string
        ProductID  string
        Quantity   int
        Amount     float64
        Reason     string // motivo de un fallo, si lo hay
        OccurredAt time.Time
}

// New crea un evento nuevo de tipo t copiando los datos de negocio de base.
func New(t Type, base Event) Event {
        base.ID = newID()
        base.Type = t
        base.OccurredAt = time.Now()
        return base
}

func newID() string {
        b := make([]byte, 6)
        _, _ = rand.Read(b)
        return hex.EncodeToString(b)
}

// Package inventory es el Inventory Service. Reserva stock cuando se crea un
// pedido y lo LIBERA (compensación) si el pago falla.
package inventory

import (
        "fmt"
        "sync"

        "saga-coreografia/internal/broker"
        "saga-coreografia/internal/events"
        "saga-coreografia/internal/logx"
)

type reservation struct {
        ProductID string
        Quantity  int
        Released  bool
}

type Service struct {
        bus broker.Broker

        mu           sync.Mutex
        stock        map[string]int          // "Inventory DB": unidades disponibles
        reservations map[string]*reservation // orderID -> reserva
        processed    map[string]bool
}

func New(bus broker.Broker, initialStock map[string]int) *Service {
        s := &Service{
                bus:          bus,
                stock:        make(map[string]int),
                reservations: make(map[string]*reservation),
                processed:    make(map[string]bool),
        }
        for k, v := range initialStock {
                s.stock[k] = v
        }
        bus.Subscribe(events.TopicOrder, "inventory-service", s.onOrderEvent)
        bus.Subscribe(events.TopicPayment, "inventory-service", s.onPaymentEvent)
        return s
}

// onOrderEvent: pasos ② y ③. Transacción local T2 = reservar stock.
func (s *Service) onOrderEvent(e events.Event) {
        if e.Type != events.OrderCreated {
                return
        }

        s.mu.Lock()
        if s.processed[e.ID] {
                s.mu.Unlock()
                return
        }
        s.processed[e.ID] = true

        available := s.stock[e.ProductID]
        if available < e.Quantity {
                s.mu.Unlock()
                reason := fmt.Sprintf("sin stock de %s (hay %d, se piden %d)", e.ProductID, available, e.Quantity)
                logx.Log("inventory", logx.Red, "② %s ✗ %s", e.OrderID, reason)
                out := events.New(events.InventoryRejected, e)
                out.Reason = reason
                s.bus.Publish(events.TopicInventory, out)
                return
        }

        s.stock[e.ProductID] = available - e.Quantity // commit local en la Inventory DB
        s.reservations[e.OrderID] = &reservation{ProductID: e.ProductID, Quantity: e.Quantity}
        left := s.stock[e.ProductID]
        s.mu.Unlock()

        logx.Log("inventory", logx.Green, "②③ %s: reservadas %d u. de %s (quedan %d) ✓", e.OrderID, e.Quantity, e.ProductID, left)
        s.bus.Publish(events.TopicInventory, events.New(events.InventoryReserved, e))
}

// onPaymentEvent: si el pago falló ejecuta la TRANSACCIÓN COMPENSATORIA de
// T2 (liberar el stock) y avisa con StockReleased.
func (s *Service) onPaymentEvent(e events.Event) {
        if e.Type != events.PaymentFailed {
                return
        }

        s.mu.Lock()
        if s.processed[e.ID] {
                s.mu.Unlock()
                return
        }
        s.processed[e.ID] = true

        r, ok := s.reservations[e.OrderID]
        if !ok || r.Released {
                s.mu.Unlock()
                return
        }
        s.stock[r.ProductID] += r.Quantity // devolvemos lo reservado
        r.Released = true
        left := s.stock[r.ProductID]
        s.mu.Unlock()

        logx.Log("inventory", logx.Yellow, "↩ COMPENSANDO %s: liberadas %d u. de %s (quedan %d)", e.OrderID, r.Quantity, r.ProductID, left)
        s.bus.Publish(events.TopicInventory, events.New(events.StockReleased, e))
}

// Stock devuelve las unidades disponibles de un producto.
func (s *Service) Stock(productID string) int {
        s.mu.Lock()
        defer s.mu.Unlock()
        return s.stock[productID]
}

// Package logx imprime mensajes con el nombre y color de cada servicio para
// poder seguir la "coreografía" en la consola.
package logx

import (
        "fmt"
        "os"
        "sync"
)

const (
        Reset  = "\033[0m"
        Blue   = "\033[34m"
        Green  = "\033[32m"
        Purple = "\033[35m"
        Yellow = "\033[33m"
        Gray   = "\033[90m"
        Red    = "\033[31m"
        Bold   = "\033[1m"
)

var (
        mu      sync.Mutex
        enabled = true
        // NO_COLOR=1 desactiva los colores (útil para guardar logs en archivos).
        noColor = os.Getenv("NO_COLOR") != ""
)

func paint(color, s string) string {
        if noColor {
                return s
        }
        return color + s + Reset
}

// SetEnabled permite silenciar los logs (lo usan los tests).
func SetEnabled(v bool) {
        mu.Lock()
        enabled = v
        mu.Unlock()
}

// Log escribe "[servicio] mensaje" con color.
func Log(service, color, format string, args ...any) {
        mu.Lock()
        defer mu.Unlock()
        if !enabled {
                return
        }
        fmt.Fprintf(os.Stdout, "%s %s\n", paint(color, fmt.Sprintf("%-11s", "["+service+"]")), fmt.Sprintf(format, args...))
}

// Title imprime un encabezado.
func Title(format string, args ...any) {
        mu.Lock()
        defer mu.Unlock()
        if !enabled {
                return
        }
        fmt.Printf("\n%s\n", paint(Bold, fmt.Sprintf(format, args...)))
}

// Package order es el Order Service. Inicia la saga y es quien decide el
// estado final del pedido (CONFIRMED o CANCELLED) según los eventos que
// escucha de Inventory y Payment.
package order

import (
        "errors"
        "fmt"
        "sync"

        "saga-coreografia/internal/broker"
        "saga-coreografia/internal/events"
        "saga-coreografia/internal/logx"
)

type Status string

const (
        Pending   Status = "PENDING"   // saga en curso
        Confirmed Status = "CONFIRMED" // todos los pasos OK
        Cancelled Status = "CANCELLED" // un paso falló y se compensó
)

type Order struct {
        ID         string
        CustomerID string
        ProductID  string
        Quantity   int
        Amount     float64
        Status     Status
        Reason     string
}

// Service tiene su PROPIA base de datos (Database-per-Service). Nadie más
// puede leerla ni escribirla.
type Service struct {
        bus broker.Broker

        mu        sync.Mutex
        db        map[string]*Order  // "Order DB"
        prices    map[string]float64 // catálogo de precios
        processed map[string]bool    // IDs de eventos ya procesados (idempotencia)
        seq       int
}

func New(bus broker.Broker, prices map[string]float64) *Service {
        s := &Service{
                bus:       bus,
                db:        make(map[string]*Order),
                prices:    prices,
                processed: make(map[string]bool),
        }
        // Coreografía: Order reacciona a lo que publican los demás.
        bus.Subscribe(events.TopicPayment, "order-service", s.onPaymentEvent)
        bus.Subscribe(events.TopicInventory, "order-service", s.onInventoryEvent)
        return s
}

// PlaceOrder es el PASO 1 de la saga (T1): transacción local que crea el
// pedido en estado PENDING y publica OrderCreated. No espera a nadie.
func (s *Service) PlaceOrder(customerID, productID string, qty int) (Order, error) {
        price, ok := s.prices[productID]
        if !ok {
                return Order{}, fmt.Errorf("producto %q no existe", productID)
        }
        if qty <= 0 {
                return Order{}, errors.New("la cantidad debe ser mayor a 0")
        }

        s.mu.Lock()
        s.seq++
        o := &Order{
                ID:         fmt.Sprintf("ORD-%03d", s.seq),
                CustomerID: customerID,
                ProductID:  productID,
                Quantity:   qty,
                Amount:     price * float64(qty),
                Status:     Pending,
        }
        s.db[o.ID] = o // commit local en la Order DB
        snapshot := *o
        s.mu.Unlock()

        logx.Log("order", logx.Blue, "① Pedido %s creado (%s x%d, $%.2f) → PENDING", o.ID, productID, qty, o.Amount)

        s.bus.Publish(events.TopicOrder, events.New(events.OrderCreated, events.Event{
                OrderID:    o.ID,
                CustomerID: customerID,
                ProductID:  productID,
                Quantity:   qty,
                Amount:     o.Amount,
        }))
        return snapshot, nil
}

// onPaymentEvent: paso ⑥ del camino feliz.
func (s *Service) onPaymentEvent(e events.Event) {
        if e.Type != events.PaymentProcessed || s.alreadyProcessed(e.ID) {
                // PaymentFailed lo ignoramos aquí: el pedido se cancela cuando
                // Inventory confirme que ya liberó el stock (StockReleased).
                return
        }
        s.setStatus(e.OrderID, Confirmed, "")
        logx.Log("order", logx.Blue, "⑥ Pago confirmado → pedido %s CONFIRMED ", e.OrderID)
}

// onInventoryEvent: aquí terminan los caminos de fallo.
func (s *Service) onInventoryEvent(e events.Event) {
        switch e.Type {
        case events.StockReleased, events.InventoryRejected:
        default:
                return
        }
        if s.alreadyProcessed(e.ID) {
                return
        }
        // Transacción COMPENSATORIA de T1: no se borra el pedido, se cancela.
        s.setStatus(e.OrderID, Cancelled, e.Reason)
        logx.Log("order", logx.Red, "↩ CancelOrder: pedido %s CANCELLED  (%s)", e.OrderID, e.Reason)
}

func (s *Service) setStatus(id string, st Status, reason string) {
        s.mu.Lock()
        defer s.mu.Unlock()
        if o, ok := s.db[id]; ok && o.Status == Pending {
                o.Status = st
                o.Reason = reason
        }
}

func (s *Service) alreadyProcessed(eventID string) bool {
        s.mu.Lock()
        defer s.mu.Unlock()
        if s.processed[eventID] {
                return true
        }
        s.processed[eventID] = true
        return false
}

// Get devuelve una copia del pedido.
func (s *Service) Get(id string) (Order, bool) {
        s.mu.Lock()
        defer s.mu.Unlock()
        o, ok := s.db[id]
        if !ok {
                return Order{}, false
        }
        return *o, true
}

// Package payment es el Payment Service. Cobra el pedido una vez que el stock
// está reservado. Es el último paso: si falla, dispara las compensaciones.
package payment

import (
        "fmt"
        "sync"

        "saga-coreografia/internal/broker"
        "saga-coreografia/internal/events"
        "saga-coreografia/internal/logx"
)

type Status string

const (
        Paid   Status = "PAID"
        Failed Status = "FAILED"
)

type Payment struct {
        OrderID    string
        CustomerID string
        Amount     float64
        Status     Status
}

type Service struct {
        bus broker.Broker

        mu        sync.Mutex
        balances  map[string]float64  // "Payment DB": saldo de cada cliente
        payments  map[string]*Payment // orderID -> pago
        processed map[string]bool
}

func New(bus broker.Broker, initialBalances map[string]float64) *Service {
        s := &Service{
                bus:       bus,
                balances:  make(map[string]float64),
                payments:  make(map[string]*Payment),
                processed: make(map[string]bool),
        }
        for k, v := range initialBalances {
                s.balances[k] = v
        }
        bus.Subscribe(events.TopicInventory, "payment-service", s.onInventoryEvent)
        return s
}

// onInventoryEvent: pasos ④ y ⑤. Transacción local T3 = cobrar.
func (s *Service) onInventoryEvent(e events.Event) {
        if e.Type != events.InventoryReserved {
                return
        }

        s.mu.Lock()
        if s.processed[e.ID] {
                s.mu.Unlock()
                return
        }
        s.processed[e.ID] = true

        balance := s.balances[e.CustomerID]
        if balance < e.Amount {
                // Se registra el intento fallido en la Payment DB.
                s.payments[e.OrderID] = &Payment{OrderID: e.OrderID, CustomerID: e.CustomerID, Amount: e.Amount, Status: Failed}
                s.mu.Unlock()

                reason := fmt.Sprintf("fondos insuficientes (saldo $%.2f, total $%.2f)", balance, e.Amount)
                logx.Log("payment", logx.Red, "④ %s ✗ pago rechazado: %s", e.OrderID, reason)
                out := events.New(events.PaymentFailed, e)
                out.Reason = reason
                s.bus.Publish(events.TopicPayment, out)
                return
        }

        s.balances[e.CustomerID] = balance - e.Amount // commit local en la Payment DB
        s.payments[e.OrderID] = &Payment{OrderID: e.OrderID, CustomerID: e.CustomerID, Amount: e.Amount, Status: Paid}
        s.mu.Unlock()

        logx.Log("payment", logx.Purple, "④⑤ %s: cobrados $%.2f a %s ✓", e.OrderID, e.Amount, e.CustomerID)
        s.bus.Publish(events.TopicPayment, events.New(events.PaymentProcessed, e))
}

// Balance devuelve el saldo de un cliente.
func (s *Service) Balance(customerID string) float64 {
        s.mu.Lock()
        defer s.mu.Unlock()
        return s.balances[customerID]
}

// Get devuelve el pago de un pedido.
func (s *Service) Get(orderID string) (Payment, bool) {
        s.mu.Lock()
        defer s.mu.Unlock()
        p, ok := s.payments[orderID]
        if !ok {
                return Payment{}, false
        }
        return *p, true
}

// Saga basada en Coreografía — caso "Black Friday" del video
// "How Saga Pattern Solves Distributed Transactions in Microservices".
//
// Tres microservicios (Order, Inventory, Payment), cada uno con su propia base
// de datos, que se coordinan SOLO mediante eventos en un broker. No existe un
// orquestador central.
package main

import (
        "fmt"

        "saga-coreografia/internal/broker"
        "saga-coreografia/internal/inventory"
        "saga-coreografia/internal/logx"
        "saga-coreografia/internal/order"
        "saga-coreografia/internal/payment"
)

// Productos de la tienda del video.
var prices = map[string]float64{
        "mystery-box": 29,
        "runner-pro":  59,
        "studio-buds": 99,
}

type system struct {
        bus       *broker.InMemory
        orders    *order.Service
        inventory *inventory.Service
        payments  *payment.Service
}

func newSystem(stock map[string]int, balances map[string]float64) *system {
        bus := broker.NewInMemory()
        return &system{
                bus:       bus,
                orders:    order.New(bus, prices),
                inventory: inventory.New(bus, stock),
                payments:  payment.New(bus, balances),
        }
}

func main() {
        sys := newSystem(
                map[string]int{"mystery-box": 100, "runner-pro": 50, "studio-buds": 3},
                map[string]float64{"Jeremy Delgado": 500, "Angel Alvarez": 20, "Erick Cevallos": 1000},
        )
        defer sys.bus.Close()

        run(sys, "ESCENARIO 1 · Camino feliz: Jeremy Delgado compra 1 Runner Pro", "Jeremy Delgado", "runner-pro", 1)
        run(sys, "ESCENARIO 2 · Falla el pago (caso Black Friday): Angel Alvarez compra 1 Studio Buds", "Angel Alvarez", "studio-buds", 1)
        run(sys, "ESCENARIO 3 · Sin stock: Erick Cevallos compra 5 Studio Buds", "Erick Cevallos", "studio-buds", 5)
}

func run(sys *system, title, customer, product string, qty int) {
        logx.Title("═══ %s ═══", title)
        stockBefore := sys.inventory.Stock(product)
        balanceBefore := sys.payments.Balance(customer)

        o, err := sys.orders.PlaceOrder(customer, product, qty)
        if err != nil {
                fmt.Println("error:", err)
                return
        }

        // La saga es ASÍNCRONA: PlaceOrder ya devolvió el pedido en PENDING.
        // Esperamos a que el broker termine de repartir todos los eventos.
        sys.bus.WaitIdle()

        final, _ := sys.orders.Get(o.ID)
        pay, hasPay := sys.payments.Get(o.ID)
        payStatus := "— (nunca llegó a cobrarse)"
        if hasPay {
                payStatus = string(pay.Status)
        }

        fmt.Printf("\n  Estado final en cada base de datos:\n")
        fmt.Printf("  • Order DB     → %s %s\n", final.ID, final.Status)
        fmt.Printf("  • Inventory DB → %s: %d → %d\n", product, stockBefore, sys.inventory.Stock(product))
        fmt.Printf("  • Payment DB   → pago %s | saldo de %s: $%.2f → $%.2f\n", payStatus, customer, balanceBefore, sys.payments.Balance(customer))
}
